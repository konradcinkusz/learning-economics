% Una etiqueta de precio: el cartón, su agujero y el precio (#1 €). En
% las unidades de las cosas, para ponerla debajo de una.
\newcommand{\etiqueta}[1]{%
  \filldraw[fill=white, line width=1pt, line join=round]
    (-0.7,-0.38) -- (0.5,-0.38) -- (0.78,0) -- (0.5,0.38) -- (-0.7,0.38) -- cycle;
  \draw[line width=0.8pt] (0.52,0) circle (0.07);
  \node[font=\fontsize{17}{17}\selectfont\bfseries] at (-0.12,0) {#1\,€};}

% Un bote de "Los tres botes": el tarro, su tapa y su nombre (#1), y
% debajo lo que hay dentro (#2: una cantidad, o \huecoRespuesta). En cm.
\newcommand{\bote}[2]{%
  \begin{tikzpicture}[line width=1.2pt, line join=round, baseline=(current bounding box.center)]
    \filldraw[fill=white, rounded corners=3mm] (-1.5,-1.8) rectangle (1.5,1.4);
    \filldraw[fill=white, rounded corners=1mm] (-1.25,1.4) rectangle (1.25,1.85);
    \node[font=\Large\bfseries] at (0,0.75) {#1};
    \node at (0,-0.55) {\fontsize{30}{36}\selectfont\bfseries #2};
  \end{tikzpicture}}
